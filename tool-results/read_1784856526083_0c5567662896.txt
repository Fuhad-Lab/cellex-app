     1→/**
     2→ * PageSkeleton — pixel-perfect loading skeletons that EXACTLY match each page's
     3→ * actual DOM structure.
     4→ *
     5→ * KEY PRINCIPLE: When the real content loads and replaces the skeleton, there
     6→ * must be ZERO layout shift. Every skeleton block has the same width, height,
     7→ * padding, margin, border-radius, and position as its real counterpart.
     8→ *
     9→ * The shimmer class is defined in globals.css.
    10→ */
    11→
    12→type Variant =
    13→  | 'home' | 'categories' | 'product' | 'cart' | 'checkout' | 'orders'
    14→  | 'search' | 'profile' | 'login' | 'ai-chat' | 'videos' | 'live'
    15→  | 'group-buy' | 'wishlist' | 'link-account' | 'telegram' | 'payment'
    16→  | 'seller-dashboard' | 'seller-products' | 'seller-orders' | 'seller-profile'
    17→  | 'seller-go-live' | 'seller-videos' | 'seller-stories' | 'seller-settings'
    18→  | 'seller-academy' | 'become-seller' | 'settings' | 'messenger'
    19→  | 'notifications' | 'create' | 'minimal';
    20→
    21→export function PageSkeleton({ variant = 'minimal' }: { variant?: Variant }) {
    22→  switch (variant) {
    23→    case 'home':              return <HomeSkeleton />;
    24→    case 'categories':        return <CategoriesSkeleton />;
    25→    case 'product':           return <ProductSkeleton />;
    26→    case 'cart':              return <CartSkeleton />;
    27→    case 'checkout':          return <CheckoutSkeleton />;
    28→    case 'orders':            return <OrdersSkeleton />;
    29→    case 'search':            return <SearchSkeleton />;
    30→    case 'profile':           return <ProfileSkeleton />;
    31→    case 'login':             return <LoginSkeleton />;
    32→    case 'ai-chat':           return <AIChatSkeleton />;
    33→    case 'videos':            return <VideosSkeleton />;
    34→    case 'live':              return <LiveSkeleton />;
    35→    case 'group-buy':         return <GroupBuySkeleton />;
    36→    case 'wishlist':          return <WishlistSkeleton />;
    37→    case 'link-account':      return <LinkAccountSkeleton />;
    38→    case 'telegram':          return <TelegramSkeleton />;
    39→    case 'payment':           return <PaymentSkeleton />;
    40→    case 'seller-dashboard':  return <SellerDashboardSkeleton />;
    41→    case 'seller-products':   return <SellerProductsSkeleton />;
    42→    case 'seller-orders':     return <SellerOrdersSkeleton />;
    43→    case 'seller-profile':    return <SellerProfileSkeleton />;
    44→    case 'seller-go-live':    return <SellerGoLiveSkeleton />;
    45→    case 'seller-videos':     return <SellerVideosSkeleton />;
    46→    case 'seller-stories':    return <SellerStoriesSkeleton />;
    47→    case 'seller-settings':   return <SellerSettingsSkeleton />;
    48→    case 'seller-academy':    return <SellerAcademySkeleton />;
    49→    case 'become-seller':     return <BecomeSellerSkeleton />;
    50→    case 'settings':          return <SettingsSkeleton />;
    51→    case 'messenger':         return <MessengerSkeleton />;
    52→    case 'notifications':     return <NotificationsSkeleton />;
    53→    case 'create':            return <CreateSkeleton />;
    54→    default:                  return <MinimalSkeleton />;
    55→  }
    56→}
