                {/* Add comment input */}
                <form onSubmit={handleAddComment} className="pt-3 border-t border-slate-800 flex gap-2">
                  <input
                    type="text"
                    placeholder="Add a comment..."
                    value={newComment}
                    onChange={(e) => setNewComment(e.target.value)}
                    className="flex-1 bg-slate-950 border border-slate-800 text-xs rounded-xl px-3 py-2.5 focus:outline-none focus:border-rose-500"
                  />
                  <button type="submit" className="bg-rose-600 text-white font-bold text-xs px-4 py-2.5 rounded-xl">
                    Post
                  </button>
                </form>
              </div>
            </div>
          )}

          {/* 6. CART BAG SLIDE-OVER */}
          {isCartOpen && (
            <div className="fixed inset-0 z-50 bg-slate-950/80 backdrop-blur-sm flex justify-end">
              <div className="w-full max-w-md bg-slate-900 h-full p-5 border-l border-slate-800 flex flex-col justify-between shadow-2xl">
                <div>
                  <div className="flex items-center justify-between mb-4 pb-3 border-b border-slate-800">
                    <span className="font-extrabold text-sm text-slate-100 flex items-center gap-2">
                      <i className="fa-solid fa-bag-shopping text-rose-500"></i> Shopping Bag ({cart.length})
                    </span>
                    <button 
                      onClick={() => setIsCartOpen(false)}
                      className="p-1.5 text-slate-400 hover:text-white rounded-full bg-slate-800"
                    >
                      <i className="fa-solid fa-xmark"></i>
                    </button>
                  </div>

                  {cart.length === 0 ? (
                    <p className="text-xs text-slate-400 text-center py-10">Your bag is currently empty.</p>
                  ) : (
                    <div className="space-y-3">
                      {cart.map((item) => (
                        <div key={item.id} className="flex items-center justify-between bg-slate-950 p-3 rounded-2xl border border-slate-800">
                          <div className="flex items-center gap-3">
                            <img src={item.media[0]} className="w-12 h-12 rounded-xl object-cover" />
                            <div>
                              <h5 className="font-bold text-xs text-slate-200 line-clamp-1">{item.title}</h5>
                              <p className="text-xs text-rose-400 font-bold">${item.price.toFixed(2)} x {item.qty}</p>
                            </div>
                          </div>
                        </div>
                      ))}
                    </div>
                  )}
                </div>

                {cart.length > 0 && (
                  <button 
                    onClick={() => { setIsCartOpen(false); setCheckoutItem(cart[0]); }}
                    className="w-full bg-rose-600 hover:bg-rose-500 text-white font-extrabold text-xs py-3.5 rounded-2xl shadow-xl transition"
                  >
                    Proceed to Checkout (${cart.reduce((a, b) => a + (b.price * b.qty), 0).toFixed(2)})
                  </button>
                )}
              </div>
            </div>
          )}

          {/* 7. NOTIFICATIONS SLIDE-OVER */}
          {isNotifOpen && (
            <div className="fixed inset-0 z-50 bg-slate-950/80 backdrop-blur-sm flex justify-end">
              <div className="w-full max-w-md bg-slate-900 h-full p-5 border-l border-slate-800 flex flex-col justify-between shadow-2xl">
                <div>
                  <div className="flex items-center justify-between mb-4 pb-3 border-b border-slate-800">
                    <span className="font-bold text-xs text-slate-200">Activity & Alerts</span>
                    <button 
                      onClick={() => setIsNotifOpen(false)}
                      className="p-1.5 text-slate-400 hover:text-white rounded-full bg-slate-800"
                    >
                      <i className="fa-solid fa-xmark"></i>
                    </button>
                  </div>

                  <div className="space-y-2 text-xs">
                    <div className="p-3 bg-slate-950 rounded-2xl border border-slate-800 flex items-center gap-3">
                      <div className="w-8 h-8 rounded-full bg-purple-600/20 text-purple-400 flex items-center justify-center">
                        <i className="fa-solid fa-users"></i>
                      </div>
                      <div>
                        <p className="font-semibold text-slate-200">Group Buy Price Unlocked!</p>
                        <p className="text-[10px] text-slate-400">@sam_dev joined your group buy for Aura Headphones.</p>
                      </div>
                    </div>

                    <div className="p-3 bg-slate-950 rounded-2xl border border-slate-800 flex items-center gap-3">
                      <div className="w-8 h-8 rounded-full bg-rose-600/20 text-rose-400 flex items-center justify-center">
                        <i className="fa-solid fa-tag"></i>
                      </div>
                      <div>
                        <p className="font-semibold text-slate-200">Price Drop Alert</p>
                        <p className="text-[10px] text-slate-400">Vintage Leather Jacket is now 32% OFF for the next hour!</p>
                      </div>
                    </div>
                  </div>
                </div>
              </div>
            </div>
          )}

        </div>
      );
    }

    const root = ReactDOM.createRoot(document.getElementById('root'));
    root.render(<App />);
  </script>
